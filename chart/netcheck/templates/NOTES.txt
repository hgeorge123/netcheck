   kubectl -n {{ include "netcheck.namespace" . }} get secret {{ include "netcheck.bootstrapSecretName" . }} -o jsonpath='{.data.{{ .Values.auth.bootstrapAdmin.secretKey }}}' | base64 -d; echo
{{- if .Values.auth.bootstrapAdmin.forcePasswordChange }}

{{- end }}

3. Open NetCheck:
{{- if .Values.httpRoute.enabled }}
{{- range .Values.httpRoute.hostnames }}
   https://{{ . }}
{{- end }}
   Check the route is accepted by the Gateway:
   kubectl -n {{ include "netcheck.namespace" . }} get httproute {{ include "netcheck.fullname" . }} -o jsonpath='{.status.parents[*].conditions}'
{{- else if .Values.ingress.enabled }}
{{- range .Values.ingress.hosts }}
   https://{{ . }}
{{- end }}
   The AWS Load Balancer Controller provisions the ALB ({{ .Values.ingress.alb.scheme }}). Its DNS name, for
